{{- /* Responsive image. Context: dict "img" "alt" "lazy" JPEG/PNG are converted to WebP with a srcset; other types (SVG, GIF, WebP) are output as is. */ -}} {{- $img := .img -}} {{- $lazy := true -}} {{- if isset . "lazy" }}{{ $lazy = .lazy }}{{ end -}} {{- if in (slice "jpeg" "png") $img.MediaType.SubType -}} {{- $srcset := slice -}} {{- $fallback := $img -}} {{- range slice 480 960 1600 -}} {{- if le . $img.Width -}} {{- $r := $img.Resize (printf "%dx webp q80" .) -}} {{- $srcset = $srcset | append (printf "%s %dw" $r.RelPermalink .) -}} {{- $fallback = $r -}} {{- end -}} {{- end -}} {{- if not $srcset -}} {{- $fallback = $img.Process "webp q80" -}} {{- end -}} {{- else -}} {{- end -}}
